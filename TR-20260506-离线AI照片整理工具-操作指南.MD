### 3.2 核心技术一：重复照片检测引擎

```swift
import Vision
import CryptoKit
import Photos

enum DuplicateType {
    case exact
    case similar(similarity: Float)
}

struct DuplicateGroup: Identifiable {
    let id = UUID()
    let type: DuplicateType
    var assets: [PHAsset]
    var bestAsset: PHAsset?
}

final class DuplicateDetector {
    private var featurePrintCache: [String: VNFeaturePrintObservation] = [:]
    private let maxCacheSize = 5000
    
    func computeHash(for asset: PHAsset) async -> String? {
        guard let imageData = await fetchOriginalImageData(for: asset) else { return nil }
        let hash = SHA256.hash(data: imageData)
        return hash.compactMap { String(format: "%02x", $0) }.joined()
    }
    
    func computeFeaturePrint(for asset: PHAsset) async -> VNFeaturePrintObservation? {
        if let cached = featurePrintCache[asset.localIdentifier] { return cached }
        
        guard let cgImage = await fetchCGImage(for: asset) else { return nil }
        let request = VNFeaturePrintObservationRequest()
        let handler = VNImageRequestHandler(cgImage: cgImage, options: [:])
        
        do {
            try handler.perform([request])
            if let observation = request.results?.first {
                if featurePrintCache.count >= maxCacheSize {
                    featurePrintCache.removeAll()
                }
                featurePrintCache[asset.localIdentifier] = observation
                return observation
            }
        } catch {
            print("Feature print error: \(error)")
        }
        return nil
    }
    
    func findDuplicates(in assets: [PHAsset]) async -> [DuplicateGroup] {
        var hashGroups: [String: [PHAsset]] = [:]
        var featurePrints: [String: VNFeaturePrintObservation] = [:]
        
        for asset in assets {
            if let hash = await computeHash(for: asset) {
                hashGroups[hash, default: []].append(asset)
            }
            if let fp = await computeFeaturePrint(for: asset) {
                featurePrints[asset.localIdentifier] = fp
            }
        }
        
        var groups: [DuplicateGroup] = []
        
        for (_, assets) in hashGroups where assets.count > 1 {
            let best = selectBestAsset(from: assets)
            groups.append(DuplicateGroup(type: .exact, assets: assets, bestAsset: best))
        }
        
        let exactDuplicateIds = Set(hashGroups.values.flatMap { $0 }.map { $0.localIdentifier })
        let remainingAssets = assets.filter { !exactDuplicateIds.contains($0.localIdentifier) }
        
        let similarGroups = await findSimilarGroups(
            assets: remainingAssets,
            featurePrints: featurePrints,
            threshold: 0.85
        )
        groups.append(contentsOf: similarGroups)
        
        return groups.sorted { $0.assets.count * calculateGroupSize($0) > $1.assets.count * calculateGroupSize($1) }
    }
    
    private func findSimilarGroups(
        assets: [PHAsset],
        featurePrints: [String: VNFeaturePrintObservation],
        threshold: Float
    ) async -> [DuplicateGroup] {
        var visited = Set<String>()
        var groups: [DuplicateGroup] = []
        
        for i in 0..<assets.count {
            let assetI = assets[i]
            guard !visited.contains(assetI.localIdentifier),
                  let fpI = featurePrints[assetI.localIdentifier] else { continue }
            
            var similarAssets = [assetI]
            
            for j in (i+1)..<assets.count {
                let assetJ = assets[j]
                guard !visited.contains(assetJ.localIdentifier),
                      let fpJ = featurePrints[assetJ.localIdentifier] else { continue }
                
                do {
                    let distance = try fpI.distance(to: fpJ)
                    let similarity = 1.0 - distance
                    
                    if similarity >= threshold {
                        similarAssets.append(assetJ)
                        visited.insert(assetJ.localIdentifier)
                    }
                } catch { continue }
            }
            
            if similarAssets.count > 1 {
                visited.insert(assetI.localIdentifier)
                let best = selectBestAsset(from: similarAssets)
                groups.append(DuplicateGroup(
                    type: .similar(similarity: threshold),
                    assets: similarAssets,
                    bestAsset: best
                ))
            }
        }
        
        return groups
    }
    
    private func selectBestAsset(from assets: [PHAsset]) -> PHAsset {
        assets.max { a, b in
            let scoreA = computeQualityScore(for: a)
            let scoreB = computeQualityScore(for: b)
            return scoreA < scoreB
        } ?? assets[0]
    }
    
    private func computeQualityScore(for asset: PHAsset) -> Double {
        var score = 0.0
        if asset.pixelWidth > 3000 { score += 3.0 }
        else if asset.pixelWidth > 2000 { score += 2.0 }
        else { score += 1.0 }
        
        if asset.mediaSubtypes.contains(.photoLive) { score += 0.5 }
        if asset.isFavorite { score += 2.0 }
        
        return score
    }
    
    private func fetchOriginalImageData(for asset: PHAsset) async -> Data? {
        let options = PHImageRequestOptions()
        options.isSynchronous = false
        options.deliveryMode = .highQualityFormat
        options.isNetworkAccessAllowed = false
        
        return await withCheckedContinuation { continuation in
            PHImageManager.default().requestImageDataAndOrientation(for: asset, options: options) { data, _, _, _ in
                continuation.resume(returning: data)
            }
        }
    }
    
    private func fetchCGImage(for asset: PHAsset) async -> CGImage? {
        let options = PHImageRequestOptions()
        options.isSynchronous = false
        options.deliveryMode = .opportunistic
        options.isNetworkAccessAllowed = false
        let targetSize = CGSize(width: 512, height: 512)
        
        return await withCheckedContinuation { continuation in
            PHImageManager.default().requestImage(
                for: asset,
                targetSize: targetSize,
                contentMode: .aspectFit,
                options: options
            ) { image, _ in
                continuation.resume(returning: image?.cgImage)
            }
        }
    }
    
    private func calculateGroupSize(_ group: DuplicateGroup) -> Int {
        group.assets.reduce(0) { $0 + Int($1.pixelWidth * $1.pixelHeight) }
    }
}
```

### 3.3 核心技术二：模糊照片检测

```swift
import Vision
import CoreImage

final class BlurDetector {
    private let context = CIContext()
    
    struct BlurResult {
        let asset: PHAsset
        let isBlurry: Bool
        let sharpnessScore: Float
    }
    
    func detectBlur(in asset: PHAsset) async -> BlurResult {
        guard let cgImage = await fetchCGImage(for: asset) else {
            return BlurResult(asset: asset, isBlurry: false, sharpnessScore: 1.0)
        }
        
        let laplacianVariance = computeLaplacianVariance(cgImage: cgImage)
        let isBlurry = laplacianVariance < 100.0
        
        return BlurResult(
            asset: asset,
            isBlurry: isBlurry,
            sharpnessScore: min(Float(laplacianVariance) / 500.0, 1.0)
        )
    }
    
    func batchDetectBlur(in assets: [PHAsset]) async -> [BlurResult] {
        await withTaskGroup(of: BlurResult.self) { group in
            for asset in assets {
                group.addTask {
                    await self.detectBlur(in: asset)
                }
            }
            
            var results: [BlurResult] = []
            for await result in group {
                results.append(result)
            }
            return results.sorted { $0.sharpnessScore < $1.sharpnessScore }
        }
    }
    
    private func computeLaplacianVariance(cgImage: CGImage) -> Double {
        let ciImage = CIImage(cgImage: cgImage)
        
        guard let grayscale = ciImage.applyingFilter("CIColorMatrix", parameters: [
            "inputRVector": CIVector(x: 0.299, y: 0, z: 0, w: 0),
            "inputGVector": CIVector(x: 0, y: 0.587, z: 0, w: 0),
            "inputBVector": CIVector(x: 0, y: 0, z: 0.114, w: 0),
            "inputAVector": CIVector(x: 0, y: 0, z: 0, w: 1),
            "inputBiasVector": CIVector(x: 0, y: 0, z: 0, w: 0)
        ]) else { return 0 }
        
        guard let laplacian = grayscale.applyingFilter("CIConvolution3X3", parameters: [
            "inputWeights": CIVector(values: [0, 1, 0, 1, -4, 1, 0, 1, 0], count: 9)
        ]) else { return 0 }
        
        guard let outputImage = context.createCGImage(laplacian, from: laplacian.extent) else { return 0 }
        
        guard let dataProvider = outputImage.dataProvider,
              let data = dataProvider.data else { return 0 }
        
        let buffer = CFDataGetBytePtr(data)!
        let length = CFDataGetLength(data)
        
        var sum: Double = 0
        for i in 0..<length {
            sum += Double(buffer[i])
        }
        let mean = sum / Double(length)
        
        var varianceSum: Double = 0
        for i in 0..<length {
            varianceSum += pow(Double(buffer[i]) - mean, 2)
        }
        let variance = varianceSum / Double(length)
        
        return variance
    }
    
    private func fetchCGImage(for asset: PHAsset) async -> CGImage? {
        let options = PHImageRequestOptions()
        options.isSynchronous = false
        options.deliveryMode = .opportunistic
        options.isNetworkAccessAllowed = false
        let targetSize = CGSize(width: 512, height: 512)
        
        return await withCheckedContinuation { continuation in
            PHImageManager.default().requestImage(
                for: asset, targetSize: targetSize, contentMode: .aspectFit, options: options
            ) { image, _ in continuation.resume(returning: image?.cgImage) }
        }
    }
}
```

```swift
final class SmartClassifier {
    private let faceDetector = FaceDetector()
    
    func classify(asset: PHAsset) async -> PhotoCategory {
        if asset.mediaType == .video { return .video }
        if asset.mediaSubtypes.contains(.photoLive) { return .livePhoto }
        if isScreenshot(asset: asset) { return .screenshot }
        
        let faceCount = await faceDetector.detectFaceCount(in: asset)
        if faceCount == 1 { return .selfie }
        
        return .normal
    }
    
    private func isScreenshot(asset: PHAsset) -> Bool {
        if asset.mediaSubtypes.contains(.screenshot) { return true }
        
        let screenScale = UIScreen.main.scale
        let screenWidth = UIScreen.main.bounds.width * screenScale
        let screenHeight = UIScreen.main.bounds.height * screenScale
        
        let aspectRatio = Double(asset.pixelWidth) / Double(asset.pixelHeight)
        let screenAspectRatio = screenWidth / screenHeight
        
        return abs(aspectRatio - screenAspectRatio) < 0.01 &&
               (asset.pixelWidth == Int(screenWidth) || asset.pixelHeight == Int(screenHeight))
    }
}

final class FaceDetector {
    func detectFaceCount(in asset: PHAsset) async -> Int {
        guard let cgImage = await fetchCGImage(for: asset) else { return 0 }
        
        let request = VNDetectFaceRectanglesRequest()
        let handler = VNImageRequestHandler(cgImage: cgImage, options: [:])
        
        do {
            try handler.perform([request])
            return request.results?.count ?? 0
        } catch {
            return 0
        }
    }
    
    func groupFaces(in assets: [PHAsset]) async -> [FaceGroup] {
        var faceObservations: [(PHAsset, [VNFaceObservation])] = []
        
        await withTaskGroup(of: (PHAsset, [VNFaceObservation]).self) { group in
            for asset in assets {
                group.addTask {
                    let faces = await self.detectFaceObservations(in: asset)
                    return (asset, faces)
                }
            }
            
            for await (asset, faces) in group {
                if !faces.isEmpty {
                    faceObservations.append((asset, faces))
                }
            }
        }
        
        return clusterFaces(faceObservations)
    }
    
    private func detectFaceObservations(in asset: PHAsset) async -> [VNFaceObservation] {
        guard let cgImage = await fetchCGImage(for: asset) else { return [] }
        let request = VNDetectFaceRectanglesRequest()
        let handler = VNImageRequestHandler(cgImage: cgImage, options: [:])
        
        do {
            try handler.perform([request])
            return request.results as? [VNFaceObservation] ?? []
        } catch {
            return []
        }
    }
    
    private func clusterFaces(_ observations: [(PHAsset, [VNFaceObservation])]) -> [FaceGroup] {
        var groups: [FaceGroup] = []
        var processed = Set<Int>()
        
        for i in 0..<observations.count {
            if processed.contains(i) { continue }
            
            var groupAssets = [observations[i].0]
            processed.insert(i)
            
            for j in (i+1)..<observations.count {
                if processed.contains(j) { continue }
                
                if facesMatch(observations[i].1, observations[j].1) {
                    groupAssets.append(observations[j].0)
                    processed.insert(j)
                }
            }
            
            if groupAssets.count > 1 {
                groups.append(FaceGroup(id: UUID(), assets: groupAssets))
            }
        }
        
        return groups
    }
    
    private func facesMatch(_ faces1: [VNFaceObservation], _ faces2: [VNFaceObservation]) -> Bool {
        guard !faces1.isEmpty && !faces2.isEmpty else { return false }
        return abs(faces1[0].boundingBox.width - faces2[0].boundingBox.width) < 0.1
    }
    
    private func fetchCGImage(for asset: PHAsset) async -> CGImage? {
        let options = PHImageRequestOptions()
        options.isSynchronous = false
        options.deliveryMode = .opportunistic
        options.isNetworkAccessAllowed = false
        let targetSize = CGSize(width: 512, height: 512)
        
        return await withCheckedContinuation { continuation in
            PHImageManager.default().requestImage(
                for: asset, targetSize: targetSize, contentMode: .aspectFit, options: options
            ) { image, _ in continuation.resume(returning: image?.cgImage) }
        }
    }
}

struct FaceGroup: Identifiable {
    let id: UUID
    var assets: [PHAsset]
}
```

```swift
import StoreKit

enum ProductID: String {
    case proUpgrade = "com.photoslim.pro"
}

@Observable
final class StoreManager {
    var isProUser = false
    var isLoading = false
    
    private var transactionListener: Task<Void, Never>?
    
    init() {
        transactionListener = listenForTransactions()
        Task { await checkPurchased() }
    }
    
    deinit {
        transactionListener?.cancel()
    }
    
    func purchasePro() async throws -> Bool {
        guard let product = try await Product.products(for: [ProductID.proUpgrade.rawValue]).first else {
            return false
        }
        
        let result = try await product.purchase()
        
        switch result {
        case .success(let verification):
            let transaction = try checkVerified(verification)
            isProUser = transaction.revocationDate == nil
            await transaction.finish()
            return isProUser
            
        case .userCancelled:
            return false
            
        case .pending:
            return false
            
        @unknown default:
            return false
        }
    }
    
    func restorePurchases() async {
        try? await AppStore.sync()
        await checkPurchased()
    }
    
    private func checkPurchased() async {
        guard let product = try? await Product.products(for: [ProductID.proUpgrade.rawValue]).first else { return }
        
        if case .verified(let transaction, _) = await product.currentTransaction {
            isProUser = transaction.revocationDate == nil
        } else {
            isProUser = false
        }
    }
    
    private func listenForTransactions() -> Task<Void, Never> {
        Task.detached { [weak self] in
            for await result in Transaction.updates {
                guard let self else { return }
                if case .verified(let transaction, _) = result {
                    await self.checkPurchased()
                    await transaction.finish()
                }
            }
        }
    }
    
    private func checkVerified<T>(_ result: VerificationResult<T>) throws -> T {
        switch result {
        case .unverified(_, let error):
            throw error
        case .verified(let safe):
            return safe
        }
    }
}
```

### 3.7 编写规则

| 规则 | 说明 |
|------|------|
| 架构模式 | MVVM + Repository Pattern，View → ViewModel → Service → Model |
| 并发模型 | 全部使用 Swift Concurrency (async/await)，禁止 DispatchQueue |
| 数据流 | 单向数据流：Service → ViewModel → View，使用 @Observable |
| 错误处理 | 使用 Result 类型 + 自定义 Error 枚举，不使用 try? |

| 测试策略 | Service 层 100% 单元测试；ViewModel 层 80% 覆盖；关键路径集成测试 |
| 命名规范 | Swift API Design Guidelines；文件名 PascalCase；方法名 camelCase |
| 最小版本 | iOS 17.0+ / macOS 14.0+（SwiftData + Observation 框架要求） |
| 代码注释 | 英文注释，遵循 Apple 文档注释规范 |
| 文件组织 | 按功能模块分组：Models/Services/Views/Utilities |

### 8.4 动画与交互设计

| 交互场景 | 动画效果 | 时长 | 技术实现 |
|----------|----------|------|----------|
| 扫描进度 | 环形进度条 + 照片飞入动画 | 持续 | SwiftUI `TimelineView` + `@Animatable` |

| 删除照片 | 照片缩小消失 + 空间释放数字跳动 | 0.25s | `transition(.scale.combined(with: .opacity))` |
| 清理完成 | 大号数字动画(0→3.2GB) + 绿色环形填充 | 1.0s | `AnimatableNumber` + `trim(from:to:)` |
| Tab切换 | 淡入淡出 | 0.2s | `.transition(.opacity)` |
| 卡片展开 | 从缩略图展开到全屏预览 | 0.35s | `matchedGeometryEffect(id:in:)` |
| 滑动删除 | 左滑露出红色删除按钮 | 即时 | `swipeActions(edge: .trailing)` |
| 存储环形图 | 环形图从0°旋转填充到目标角度 | 0.8s | `rotationEffect` + `trim` |

## 十一、技术风险与应对方案

| 风险 | 概率 | 影响 | 应对方案 |
|------|------|------|----------|
| Vision FeaturePrint在大图上慢 | 中 | 高 | 缩放到512px再计算；TaskGroup限制4并发 |
| 10万+照片内存溢出 | 中 | 高 | 分批处理(1000张/批)；CGImage用完立即释放；FeaturePrint缓存限制5000条 |
| App Store审核拒绝(照片删除) | 低 | 极高 | 使用PHAssetChangeRequest标准API；删除前二次确认；30天恢复期说明 |
| 相似度阈值不准(误判) | 中 | 中 | 默认85%阈值；提供滑块调节；保护收藏照片 |
| Laplacian方差对景深模糊不准 | 中 | 低 | 添加Vision VNClassifyImageRequest辅助判断；允许用户调整阈值 |
| StoreKit 2购买验证失败 | 低 | 中 | 实现Transaction.updates监听；提供恢复购买按钮；本地receipt缓存 |
| iOS 17 SwiftData迁移问题 | 低 | 中 | 使用轻量迁移(Lightweight Migration)；版本化Schema |
